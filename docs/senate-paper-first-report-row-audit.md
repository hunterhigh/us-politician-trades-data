```powershell
$env:PYTHONPATH='backend/src'
python backend/scripts/audit_senate_paper_first_report.py --repo . --inventory docs/senate-paper-52-page-inventory.json --output docs/senate-paper-first-report-rows.json
python -m unittest discover -s backend/tests -p 'test_senate_paper*.py' -v
```
